MariaDB [project_ARMS]> -- Procedure 3: Get Student Marks for Course
MariaDB [project_ARMS]> CREATE PROCEDURE GetStudentMarksForCourse(
    ->     IN p_reg_no VARCHAR(20),
    ->     IN p_course_code VARCHAR(20)
    -> )
    -> BEGIN
    ->     SELECT
    ->         a.Assignment_name,
    ->         a.type,
    ->         a.max_marks,
    ->         sm.mark_obtained,
    ->         sm.is_medical,
    ->         ROUND((sm.mark_obtained / a.max_marks) * 100, 2) AS percentage
    ->     FROM Assignment a
    ->     LEFT JOIN Student_Marks sm
    ->         ON a.Assignment_id = sm.Assignment_id AND sm.reg_no = p_reg_no
    ->     WHERE a.course_code = p_course_code
    ->     ORDER BY a.Assignment_id;
    -> END $$

MariaDB [project_ARMS]> -- Procedure 4: Calculate Total CA Marks
MariaDB [project_ARMS]> CREATE PROCEDURE CalculateTotalCAMarks(
    ->     IN p_reg_no VARCHAR(20),
    ->     IN p_course_code VARCHAR(20)
    -> )
    -> BEGIN
    ->     SELECT
    ->         p_reg_no AS reg_no,
    ->         p_course_code AS course_code,
    ->         SUM(sm.mark_obtained) AS total_marks_obtained,
    ->         SUM(a.max_marks) AS total_max_marks,
    ->         ROUND((SUM(sm.mark_obtained) / SUM(a.max_marks)) * 100, 2) AS ca_percentage
    ->     FROM Assignment a
    ->     LEFT JOIN Student_Marks sm
    ->         ON a.Assignment_id = sm.Assignment_id AND sm.reg_no = p_reg_no
    ->     WHERE a.course_code = p_course_code;
    -> END $$
